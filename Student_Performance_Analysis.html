Research Framework

Research Objectives & Dataset Anatomy

Systematically engineered according to the UCI Machine Learning repository specifications by Paulo Cortez and Alice Silva (2008).

1. Ingestion & Validation

Automated ingestion of semicolon-delimited records (student-mat.csv), type casting grade structures, verifying ordinal ranges (1-5 scales for alcohol, health, family quality), and null validation.

2. Exploratory Analytics

In-depth statistical distribution analysis highlighting grade skewness, demographic balance (52.7% Female vs 47.3% Male), and urban vs rural household dispersion.

3. Hypothesis Verification

Rigorous statistical inference executing Welch two-sample t-tests, one-way Analysis of Variance (ANOVA), and Pearson correlation hypothesis testing with α = 0.05.

4. Predictive Modeling

Formulation of an ordinary least squares (OLS) multiple linear regression model estimating the final grade G3 from historical trimesters, past failures, attendance, and study habits.

Data Cleaning & Quality Validation

Runs null checks, duplicate scans, type casting for grade columns (G1, G2, G3), and boundary validation across all 33 variables.

# =============================================================================
# 02_cleaning.R
# Purpose: Data cleaning and validation for Student Performance Dataset
# Input: students (from 01_import.R)
# Output: students_clean (validated and cleaned dataset)
# =============================================================================

# --- Source the import script --------------------------------------------------
if (!exists("students")) {
  source("R/01_import.R")
}

cat("========== DATA CLEANING REPORT ==========\n\n")

# --- 1. Missing values --------------------------------------------------------
cat("--- Missing Values Check ---\n")
missing_counts <- colSums(is.na(students))
print(missing_counts)
cat("Total missing values:", sum(is.na(students)), "\n\n")

# --- 2. Duplicate rows --------------------------------------------------------
cat("--- Duplicate Rows Check ---\n")
duplicate_count <- sum(duplicated(students))
cat("Duplicate rows:", duplicate_count, "\n\n")

# --- 3. Data types check ------------------------------------------------------
cat("--- Data Types ---\n")
print(sapply(students, class))
cat("\n")

# --- 4. Check for empty strings (common in CSV) ------------------------------
cat("--- Empty String Check ---\n")
empty_strings <- sapply(students, function(x) {
  if (is.character(x)) {
    sum(x == "" | x == " " | x == "\"\"")
  } else {
    0
  }
})
print(empty_strings[empty_strings > 0])
if (all(empty_strings == 0)) {
  cat("No empty strings found.\n")
}
cat("\n")

# --- 5. Validate numeric columns ---------------------------------------------
cat("--- Numeric Column Validation ---\n")

# Check G1, G2, G3 range (should be 0-20)
grade_cols <- c("G1", "G2", "G3")
for (col in grade_cols) {
  vals <- students[[col]]
  cat(col, " - Range:", min(vals), "to", max(vals),
      " | Any outside 0-20:", any(vals < 0 | vals > 20), "\n")
}

# Check age range (should be 15-22)
cat("age  - Range:", min(students$age), "to", max(students$age),
    " | Any outside 15-22:", any(students$age < 15 | students$age > 22), "\n")

# Check absences range (should be 0-93)
cat("absences - Range:", min(students$absences), "to", max(students$absences),
    " | Any negative:", any(students$absences < 0), "\n")

# Check ordinal variables (1-5 scale)
ordinal_cols <- c("famrel", "freetime", "goout", "Dalc", "Walc", "health")
for (col in ordinal_cols) {
  vals <- students[[col]]
  cat(col, " - Unique values:", sort(unique(vals)), "\n")
}

# Check failures range (0-4)
cat("failures - Unique values:", sort(unique(students$failures)), "\n")
cat("studytime - Unique values:", sort(unique(students$studytime)), "\n\n")

# --- 6. Validate categorical columns -----------------------------------------
cat("--- Categorical Column Validation ---\n")
cat("school  - Values:", unique(students$school), "\n")
cat("sex     - Values:", unique(students$sex), "\n")
cat("address - Values:", unique(students$address), "\n")
cat("famsize - Values:", unique(students$famsize), "\n")
cat("Pstatus - Values:", unique(students$Pstatus), "\n")
cat("Mjob    - Values:", unique(students$Mjob), "\n")
cat("Fjob    - Values:", unique(students$Fjob), "\n")
cat("reason  - Values:", unique(students$reason), "\n")
cat("guardian- Values:", unique(students$guardian), "\n")

# yes/no columns
yesno_cols <- c("schoolsup", "famsup", "paid", "activities",
                "nursery", "higher", "internet", "romantic")
for (col in yesno_cols) {
  vals <- unique(students[[col]])
  cat(col, " - Values:", vals, "\n")
}
cat("\n")

# --- 7. Create cleaned dataset ------------------------------------------------
# The UCI Student Performance Dataset is clean by design.
# We create students_clean as a working copy to preserve the original.

students_clean <- students

# Ensure grade columns are numeric (they may be read as character if quoted)
for (col in grade_cols) {
  if (!is.numeric(students_clean[[col]])) {
    students_clean[[col]] <- as.numeric(students_clean[[col]])
    cat("Converted", col, "to numeric.\n")
  }
}

# --- 8. Cleaning summary -----------------------------------------------------
cat("\n========== CLEANING SUMMARY ==========\n")
cat("Original rows   :", nrow(students), "\n")
cat("Rows after clean:", nrow(students_clean), "\n")
cat("Columns         :", ncol(students_clean), "\n")
cat("Missing values  :", sum(is.na(students_clean)), "\n")
cat("Duplicate rows  :", sum(duplicated(students_clean)), "\n")
cat("Dataset is ready for analysis.\n")

# =============================================================================

  source("R/02_cleaning.R")

cat("========== DESCRIPTIVE STATISTICS ==========\n\n")

# --- Helper function ----------------------------------------------------------
describe_numeric <- function(x, var_name) {
  cat(sprintf("--- %s ---\n", var_name))
  cat(sprintf("  Count  : %d\n", length(x)))
  cat(sprintf("  Mean   : %.2f\n", mean(x)))
  cat(sprintf("  Median : %.2f\n", median(x)))
  cat(sprintf("  Std Dev: %.2f\n", sd(x)))
  cat(sprintf("  Min    : %d\n", min(x)))
  cat(sprintf("  Max    : %d\n", max(x)))
  cat(sprintf("  Range  : %d\n", max(x) - min(x)))
  cat(sprintf("  Q1     : %.2f\n", quantile(x, 0.25)))
  cat(sprintf("  Q3     : %.2f\n", quantile(x, 0.75)))
  cat(sprintf("  IQR    : %.2f\n", IQR(x)))
  cat("\n")
}

# --- 2. Interpret correlations ------------------------------------------------
interpret_cor <- function(r) {
  abs_r <- abs(r)
  direction <- ifelse(r >= 0, "positive", "negative")
  strength <- ifelse(abs_r >= 0.7, "strong",
               ifelse(abs_r >= 0.4, "moderate",
                ifelse(abs_r >= 0.2, "weak", "very weak")))
  sprintf("%s %s correlation (r = %.4f)", strength, direction, r)
}

Hypothesis Testing & Statistical Inference

Conducts two-sample Welch t-test (gender), One-Way ANOVA (study time), Pearson correlation tests, and simple linear regression.

# =============================================================================
# 06_statistical_analysis.R
# Purpose: Statistical hypothesis testing for Student Performance Dataset
# Input: students_clean (from 02_cleaning.R)
# Tests: t-test, ANOVA, correlation test, simple linear regression
# =============================================================================

# --- Source the cleaning script ------------------------------------------------
if (!exists("students_clean")) {
  source("R/02_cleaning.R")
}

cat("========== STATISTICAL ANALYSIS ==========\n\n")

alpha <- 0.05

# =============================================================================
# TEST 1: Independent Samples t-test - Final Grade by Gender
# =============================================================================
cat("============================================\n")
cat("TEST 1: t-test - G3 by Gender (sex)\n")
cat("============================================\n\n")

cat("H0: There is no significant difference in mean final grades between male and female students.\n")
cat("H1: There is a significant difference in mean final grades between male and female students.\n")
cat(sprintf("Significance level (alpha): %.2f\n\n", alpha))

t_result <- t.test(G3 ~ sex, data = students_clean, var.equal = FALSE)
print(t_result)

cat("\n--- Interpretation ---\n")
if (t_result$p.value < alpha) {
  cat(sprintf("p-value = %.6f < %.2f: REJECT H0.\n", t_result$p.value, alpha))
  cat("There IS a statistically significant difference in final grades between genders.\n")
  cat(sprintf("Female mean: %.2f | Male mean: %.2f\n",
              mean(students_clean$G3[students_clean$sex == "F"]),
              mean(students_clean$G3[students_clean$sex == "M"])))
} else {
  cat(sprintf("p-value = %.6f >= %.2f: FAIL TO REJECT H0.\n", t_result$p.value, alpha))
  cat("There is NO statistically significant difference in final grades between genders.\n")
}

# =============================================================================
# TEST 2: One-way ANOVA - Final Grade by Study Time
# =============================================================================
cat("\n\n============================================\n")
cat("TEST 2: ANOVA - G3 by Study Time\n")
cat("============================================\n\n")

cat("H0: There is no significant difference in mean final grades across study time groups.\n")
cat("H1: At least one study time group has a significantly different mean final grade.\n")
cat(sprintf("Significance level (alpha): %.2f\n\n", alpha))

anova_result <- aov(G3 ~ factor(studytime), data = students_clean)
print(summary(anova_result))

# Post-hoc Tukey HSD if ANOVA is significant
anova_p <- summary(anova_result)[[1]]$`Pr(>F)`[1]
cat(sprintf("\nANOVA p-value: %.6f\n", anova_p))

cat("\n--- Interpretation ---\n")
if (anova_p < alpha) {
  cat(sprintf("p-value = %.6f < %.2f: REJECT H0.\n", anova_p, alpha))
  cat("Study time groups have significantly different mean final grades.\n\n")
  cat("--- Post-hoc Tukey HSD Test ---\n")
  tukey_result <- TukeyHSD(anova_result)
  print(tukey_result)
} else {
  cat(sprintf("p-value = %.6f >= %.2f: FAIL TO REJECT H0.\n", anova_p, alpha))
  cat("There is NO significant difference in final grades across study time groups.\n")
}

# =============================================================================
# TEST 3: Correlation Test - G2 vs G3
# =============================================================================
cat("\n\n============================================\n")
cat("TEST 3: Correlation Test - G2 vs G3\n")
cat("============================================\n\n")

cat("H0: There is no significant linear correlation between G2 and G3.\n")
cat("H1: There is a significant linear correlation between G2 and G3.\n")
cat(sprintf("Significance level (alpha): %.2f\n\n", alpha))

cor_test_result <- cor.test(students_clean$G2, students_clean$G3, method = "pearson")
print(cor_test_result)

cat("\n--- Interpretation ---\n")
if (cor_test_result$p.value < alpha) {
  cat(sprintf("p-value = %.6f < %.2f: REJECT H0.\n", cor_test_result$p.value, alpha))
  cat(sprintf("There IS a significant correlation between G2 and G3 (r = %.4f).\n",
              cor_test_result$estimate))
} else {
  cat(sprintf("p-value = %.6f >= %.2f: FAIL TO REJECT H0.\n", cor_test_result$p.value, alpha))
  cat("There is NO significant correlation between G2 and G3.\n")
}

# =============================================================================
# TEST 4: Correlation Test - G1 vs G3
# =============================================================================
cat("\n\n============================================\n")
cat("TEST 4: Correlation Test - G1 vs G3\n")
cat("============================================\n\n")

cat("H0: There is no significant linear correlation between G1 and G3.\n")
cat("H1: There is a significant linear correlation between G1 and G3.\n")
cat(sprintf("Significance level (alpha): %.2f\n\n", alpha))

cor_test_g1 <- cor.test(students_clean$G1, students_clean$G3, method = "pearson")
print(cor_test_g1)

cat("\n--- Interpretation ---\n")
if (cor_test_g1$p.value < alpha) {
  cat(sprintf("p-value = %.6f < %.2f: REJECT H0.\n", cor_test_g1$p.value, alpha))
  cat(sprintf("There IS a significant correlation between G1 and G3 (r = %.4f).\n",
              cor_test_g1$estimate))
} else {
  cat(sprintf("p-value = %.6f >= %.2f: FAIL TO REJECT H0.\n", cor_test_g1$p.value, alpha))
  cat("There is NO significant correlation between G1 and G3.\n")
}

# =============================================================================
# TEST 5: Simple Linear Regression - G3 predicted by G2
# =============================================================================
cat("\n\n============================================\n")
cat("TEST 5: Simple Linear Regression - G3 ~ G2\n")
cat("============================================\n\n")

cat("H0: The slope coefficient for G2 is zero (G2 does not predict G3).\n")
cat("H1: The slope coefficient for G2 is not zero (G2 significantly predicts G3).\n")
cat(sprintf("Significance level (alpha): %.2f\n\n", alpha))

simple_model <- lm(G3 ~ G2, data = students_clean)
print(summary(simple_model))

cat("\n--- Interpretation ---\n")
coefs <- coefficients(simple_model)
cat(sprintf("Intercept : %.4f\n", coefs[1]))
cat(sprintf("G2 coeff  : %.4f\n", coefs[2]))
cat(sprintf("R-squared : %.4f\n", summary(simple_model)$r.squared))
cat(sprintf("Adj R-sq  : %.4f\n", summary(simple_model)$adj.r.squared))

if (summary(simple_model)$coefficients[2, 4] < alpha) {
  cat("G2 is a statistically significant predictor of G3.\n")
} else {
  cat("G2 is NOT a statistically significant predictor of G3.\n")
}

cat("\n========== END STATISTICAL ANALYSIS ==========\n")

Multiple Linear Regression & Residual Diagnostics

Trains the multiple linear regression model (G3 ~ G1 + G2 + studytime + failures + absences), extracts R² (82.9%), and plots 4-panel residual diagnostics.

# =============================================================================
# 07_prediction.R
# Purpose: Multiple Linear Regression model for predicting Final Grade (G3)
# Input: students_clean (from 02_cleaning.R)
# Model: G3 ~ G1 + G2 + studytime + failures + absences
# =============================================================================

# --- Load required packages ---------------------------------------------------
library(ggplot2)

# --- Source the cleaning script ------------------------------------------------
if (!exists("students_clean")) {
  source("R/02_cleaning.R")
}

cat("========== PREDICTION MODEL ==========\n\n")

# --- 1. Build the Multiple Linear Regression Model ----------------------------
cat("--- Model: G3 ~ G1 + G2 + studytime + failures + absences ---\n\n")

model <- lm(G3 ~ G1 + G2 + studytime + failures + absences, data = students_clean)

# --- 2. Model Summary ---------------------------------------------------------
cat("--- Full Model Summary ---\n\n")
print(summary(model))

# --- 3. Extract key metrics ---------------------------------------------------
model_summary <- summary(model)
cat("\n--- Key Model Metrics ---\n")
cat(sprintf("R-squared       : %.4f\n", model_summary$r.squared))
cat(sprintf("Adjusted R-squared: %.4f\n", model_summary$adj.r.squared))
cat(sprintf("Residual Std Error: %.4f on %d degrees of freedom\n",
            model_summary$sigma, model_summary$df[2]))
cat(sprintf("F-statistic     : %.4f on %d and %d DF, p-value: %.2e\n",
            model_summary$fstatistic[1],
            model_summary$fstatistic[2],
            model_summary$fstatistic[3],
            pf(model_summary$fstatistic[1],
               model_summary$fstatistic[2],
               model_summary$fstatistic[3],
               lower.tail = FALSE)))

# --- 4. Coefficients table ----------------------------------------------------
cat("\n--- Coefficients ---\n")
coefs <- as.data.frame(summary(model)$coefficients)
coefs$Significant <- coefs$`Pr(>|t|)` < 0.05
print(round(coefs, 4))

# --- 5. Interpretation --------------------------------------------------------
cat("\n--- Model Interpretation (Academic Language) ---\n\n")

cat("The multiple linear regression model was built to predict the final grade (G3)\n")
cat("using five predictor variables: first period grade (G1), second period grade (G2),\n")
cat("weekly study time (studytime), past failures (failures), and absences.\n\n")

cat(sprintf("1. R-squared = %.4f means the model explains %.1f%% of the variance in G3.\n",
            model_summary$r.squared, model_summary$r.squared * 100))

cat(sprintf("2. Adjusted R-squared = %.4f accounts for the number of predictors.\n\n",
            model_summary$adj.r.squared))

cat("3. Coefficient interpretations:\n")
for (v in rownames(coefs)) {
  if (v != "(Intercept)") {
    direction <- ifelse(coefs[v, 1] > 0, "increases", "decreases")
    sig <- ifelse(coefs[v, "Significant"], "(significant)", "(not significant)")
    cat(sprintf("   - %s: A one-unit increase in %s %s G3 by %.4f points %s\n",
                v, v, direction, abs(coefs[v, 1]), sig))
  }
}

# --- 6. Residual Analysis -----------------------------------------------------
cat("\n--- Residual Analysis ---\n")

# Basic residual statistics
residuals_vals <- residuals(model)
cat("Residual summary:\n")
print(summary(residuals_vals))

# --- 7. Residual Plots --------------------------------------------------------
par(mfrow = c(2, 2))
png("plots/residual_analysis.png", width = 1000, height = 800, res = 150)
par(mfrow = c(2, 2))
plot(model, main = "Residual Analysis - Multiple Linear Regression")
dev.off()
par(mfrow = c(1, 1))
cat("\nResidual plots saved: plots/residual_analysis.png\n")

# --- 8. Residuals vs Fitted scatter (ggplot2) ---------------------------------
students_clean$fitted_values <- fitted(model)
students_clean$residuals_vals <- residuals(model)

p_resid <- ggplot(students_clean, aes(x = fitted_values, y = residuals_vals)) +
  geom_point(alpha = 0.5, color = "steelblue") +
  geom_hline(yintercept = 0, color = "red", linetype = "dashed") +
  geom_smooth(method = "loess", se = FALSE, color = "darkorange", linewidth = 0.8) +
  labs(
    title = "Residuals vs Fitted Values",
    x     = "Fitted Values",
    y     = "Residuals"
  ) +
  theme_minimal(base_size = 12) +
  theme(plot.title = element_text(face = "bold"))

ggsave("plots/residuals_vs_fitted.png", p_resid, width = 8, height = 5, dpi = 150)
cat("Residuals vs Fitted plot saved: plots/residuals_vs_fitted.png\n")

# --- 9. Model Limitations -----------------------------------------------------
cat("\n--- Model Limitations ---\n\n")
cat("1. This model is based on a single dataset from two Portuguese schools.\n")
cat("   Results may not generalize to other educational contexts.\n\n")
cat("2. G1 and G2 are highly correlated with G3 (they are from the same grading\n")
cat("   system). This makes the model strong but somewhat circular.\n\n")
